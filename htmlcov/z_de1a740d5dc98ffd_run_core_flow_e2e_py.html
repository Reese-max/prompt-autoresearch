117def _strip_volatile(summary: dict[str, Any], results: list[dict[str, Any]]) -> tuple[dict, list]: 

118 """移除時間戳等不可跨跑比對的欄位，保留契約欄位。""" 

119 summary_keys = ( 

120 "prompt_file", 

121 "question_file", 

122 "prompt_hash", 

123 "total_questions", 

124 "average_score", 

125 "word_count_pass_rate", 

126 "risk_perfect_rate", 

127 "error_count", 

128 ) 

129 result_keys = ( 

130 "id", 

131 "type", 

132 "question", 

133 "answer", 

134 "char_count", 

135 "prompt_hash", 

136 "question_file", 

137 "general_score", 

138 "type_specific_score", 

139 "risk_score", 

140 "total_score", 

141 "failures", 

142 "critique", 

143 ) 

144 clean_summary = {k: summary[k] for k in summary_keys if k in summary} 

145 clean_results = [] 

146 for row in results: 

147 clean = {k: row[k] for k in result_keys if k in row} 

148 # 路徑只保留可移植相對段，避免 tmp 絕對路徑破壞比對 

149 if "question_file" in clean: 

150 clean["question_file"] = _portable_path(clean["question_file"]) 

151 clean_results.append(clean) 

152 if "prompt_file" in clean_summary: 

153 clean_summary["prompt_file"] = _portable_path(clean_summary["prompt_file"]) 

154 if "question_file" in clean_summary: 

155 clean_summary["question_file"] = _portable_path(clean_summary["question_file"]) 

156 return clean_summary, clean_results

159def _portable_path(path_str: str) -> str: 

160 """將絕對/平台路徑收斂為以『輸入 資料/…』結尾的可移植鍵。""" 

161 normalized = io.normalize_path(path_str) 

162 marker = "輸入 資料/" 

163 idx = normalized.find(marker) 

164 if idx >= 0: 

165 return normalized[idx:] 

166 return Path(normalized).name

185 def fake_minimax(_system, _user, temperature=0.7): 

186 if temperature == 0.3: 

187 return ANSWER 

188 if temperature == 0.1: 

189 return JUDGE_OUTPUT 

190 raise AssertionError(f"unexpected temperature={temperature}") 

191 

192 original = evaluate.call_minimax 

193 evaluate.call_minimax = fake_minimax 

194 try: 

195 prompt_arg = io.normalize_path(str(prompt_path)) 

196 question_arg = io.normalize_path(str(question_path)) 

197 summary, results = evaluate.run_evaluation( 

198 prompt_arg, question_arg, max_workers=1 

199 ) 

200 # 清 cache 後再跑一次，驗證冪等 

201 cache_dir = workspace / ".cache" 

202 if cache_dir.exists(): 

203 import shutil

205 shutil.rmtree(cache_dir) 

206 summary2, results2 = evaluate.run_evaluation( 

207 prompt_arg, question_arg, max_workers=1 

208 ) 

209 run_dir = Path(evaluate.save_run_results(summary, results)) 

210 finally: 

211 evaluate.call_minimax = original

213 artifacts = {} 

214 for name in ("details.jsonl", "summary.json", "summary.md"):

216 artifacts[name] = { 

217 "exists": True, 

218 "has_bom": raw.startswith(b"\xef\xbb\xbf"), 

219 "sha256": _sha256_bytes(raw), 

220 "utf8_ok": True, 

221 } 

222 raw.decode("utf-8") 

223 

224 tsv_raw = Path("results.tsv").read_bytes() 

225 # TSV 第一欄常為 run id（時間戳），正規化為只比對分數欄位列 

226 tsv_text = tsv_raw.decode("utf-8") 

227 tsv_score_marker = "\t79.00\t" in tsv_text 

228 artifacts["results.tsv"] = { 

229 "exists": True, 

230 "has_bom": tsv_raw.startswith(b"\xef\xbb\xbf"), 

231 "contains_score_79": tsv_score_marker, 

232 "sha256_full": _sha256_bytes(tsv_raw), 

233 }

235 clean_summary, clean_results = _strip_volatile(summary, results) 

236 clean_summary2, clean_results2 = _strip_volatile(summary2, results2)

238 return { 

239 "summary": clean_summary, 

240 "results": clean_results, 

241 "repeat_identical": ( 

242 clean_summary == clean_summary2 and clean_results == clean_results2 

243 ), 

244 "artifacts": artifacts, 

245 "latest_symlink_ok": all( 

246 (Path("runs/latest") / n).exists() 

247 for n in ("details.jsonl", "summary.json", "summary.md")

249 "canonical_digest": _sha256_text(

276def _check_spec(eval_result: dict[str, Any], gate: dict[str, Any]) -> list[str]: 

277 failures: list[str] = [] 

278 s = eval_result["summary"] 

279 r0 = eval_result["results"][0] if eval_result["results"] else {} 

280 

281 checks = [ 

282 (s.get("prompt_hash") == EXPECTED["prompt_hash"], "prompt_hash"), 

283 (s.get("total_questions") == EXPECTED["total_questions"], "total_questions"), 

284 (s.get("average_score") == EXPECTED["average_score"], "average_score"), 

285 ( 

286 s.get("word_count_pass_rate") == EXPECTED["word_count_pass_rate"], 

287 "word_count_pass_rate", 

288 ), 

289 (s.get("risk_perfect_rate") == EXPECTED["risk_perfect_rate"], "risk_perfect_rate"), 

290 (s.get("error_count") == EXPECTED["error_count"], "error_count"), 

291 (r0.get("total_score") == EXPECTED["total_score"], "total_score"), 

292 (r0.get("failures") == EXPECTED["failures"], "failures"), 

293 (r0.get("char_count") == EXPECTED["char_count"], "char_count"), 

294 (r0.get("answer") == ANSWER, "answer"), 

295 (eval_result.get("repeat_identical") is True, "repeat_identical"), 

296 (eval_result.get("latest_symlink_ok") is True, "latest_symlink_ok"), 

297 ( 

298 all(not a.get("has_bom") for a in eval_result["artifacts"].values() if "has_bom" in a), 

299 "no_bom", 

300 ), 

301 ( 

302 eval_result["artifacts"]["results.tsv"].get("contains_score_79") is True, 

303 "tsv_score", 

304 ), 

305 ( 

306 gate["valid_prompt"]["passed"] is EXPECTED["gatekeeper_passed"], 

307 "gatekeeper_passed", 

308 ), 

309 ( 

310 gate["valid_prompt"]["char_count"] == EXPECTED["gatekeeper_char_count"], 

311 "gatekeeper_char_count", 

312 ), 

313 (gate["short_prompt"]["passed"] is False, "gatekeeper_reject_short"), 

314 (gate["short_prompt"]["reject_count"] >= 1, "gatekeeper_reject_count"), 

315 ] 

316 for ok, name in checks: 

317 if not ok: 

318 failures.append(name) 

319 return failures

322def run_e2e() -> dict[str, Any]: 

323 runtime = _runtime_fingerprint() 

324 old_cwd = Path.cwd() 

325 with tempfile.TemporaryDirectory(prefix="core-flow-e2e-") as tmp: 

326 workspace = Path(tmp) 

327 os.chdir(workspace) 

328 try: 

329 eval_result = _run_core_evaluation(workspace) 

330 gate = _run_gatekeeper_probe() 

331 finally: 

332 os.chdir(old_cwd) 

333 

334 spec_failures = _check_spec(eval_result, gate) 

335 comparable = { 

336 "schema_version": 1, 

337 "flow": "core_evaluation_e2e", 

338 "expected": EXPECTED, 

339 "gatekeeper": gate, 

340 "evaluation": { 

341 "summary": eval_result["summary"], 

342 "results": eval_result["results"], 

343 "repeat_identical": eval_result["repeat_identical"], 

344 "latest_symlink_ok": eval_result["latest_symlink_ok"], 

345 "canonical_digest": eval_result["canonical_digest"], 

346 "artifact_flags": { 

347 name: { 

348 k: v 

349 for k, v in meta.items() 

350 if k in ("exists", "has_bom", "contains_score_79", "utf8_ok") 

351 } 

352 for name, meta in eval_result["artifacts"].items() 

353 }, 

354 }, 

355 } 

356 # 完整可比對 digest：不含 runtime 指紋 

357 comparable_digest = _sha256_text(_canonical_json(comparable)) 

358 

359 return { 

360 "runtime": runtime, 

361 "comparable": comparable, 

362 "comparable_digest": comparable_digest, 

363 "spec_ok": len(spec_failures) == 0, 

364 "spec_failures": spec_failures, 

365 "exit_code": 0 if not spec_failures else 1, 

366 }

369def main(argv: list[str] | None = None) -> int: 

370 parser = argparse.ArgumentParser(description="核心流程端到端測試（可比對輸出）") 

371 parser.add_argument( 

372 "--out", 

373 type=str,

376 )

378 "--quiet", 

379 action="store_true", 

380 help="只輸出一行 SUMMARY JSON", 

381 ) 

382 args = parser.parse_args(argv) 

383 

384 result = run_e2e() 

385 payload = { 

386 "runtime": result["runtime"], 

387 "comparable_digest": result["comparable_digest"], 

388 "spec_ok": result["spec_ok"], 

389 "spec_failures": result["spec_failures"], 

390 "comparable": result["comparable"], 

391 "exit_code": result["exit_code"], 

392 } 

393 out_path = None 

394 out_parent_created = False 

395 

396 try: 

397 if args.out: 

398 out_path = Path(args.out) 

399 if not out_path.is_absolute(): 

400 out_path = PROJECT_ROOT / out_path

402 out_path.parent.mkdir(parents=True, exist_ok=True) 

403 out_parent_created = True 

404 out_path.write_text( 

405 json.dumps(payload, ensure_ascii=False, indent=2) + "\n",

407 ) 

408 

409 summary_line = { 

410 "platform": result["runtime"]["platform"], 

411 "python_version": result["runtime"]["python_version"], 

412 "comparable_digest": result["comparable_digest"], 

413 "spec_ok": result["spec_ok"],

428 except Exception: 

429 if out_path is not None and out_path.exists():

432 try:
