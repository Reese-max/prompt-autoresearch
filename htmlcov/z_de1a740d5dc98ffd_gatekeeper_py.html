3scripts/gatekeeper.py — 硬性規則防呆檢查器 

4檢查 prompt 是否違反硬性規則，不過就淘汰，不進入正式測試。 

5用法：python3 scripts/gatekeeper.py prompts/current.md 

6"""

21# 語意等價詞表，用於第二層語意驗證 (#15) 

22SEMANTIC_EQUIVALENTS = { 

23 "R05_NO_EXPERT_ROLE": [ 

24 ["專家", "閱卷委員", "評分委員", "寫作名師", "考官", "老師", "命題老師", "國考顧問", "考試專家"], 

25 ], 

26 "R03_NO_ANTI_QUESTION": [ 

27 ["不得反問", "禁止反問", "不要反問", "不得提問", "不得摻雜.*提問", "禁止.*提問", "不得詢問", "禁止詢問"], 

28 ], 

29 "R04_NO_ANTI_FABRICATION": [ 

30 ["不得編造", "嚴禁編造", "嚴禁捏造", "不要編造", "禁止編造", "杜絕捏造", "嚴禁憑空捏造", "不得.*捏造", "不可編造", "不可捏造"], 

31 ], 

32}

35def _semantic_check(rule, prompt_text): 

36 """第二層語意驗證：檢查 prompt 是否包含語意等價詞。""" 

37 equivalents = SEMANTIC_EQUIVALENTS.get(rule) 

38 if not equivalents: 

39 return False # 無等價詞表，無法語意驗證 

40 for group in equivalents: 

41 if any(re.search(keyword, prompt_text) for keyword in group): 

42 return True 

43 return False

46def run_gatekeeper(prompt_text): 

47 """ 

48 對提示詞進行硬性規則檢查。 

49 回傳 (passed: bool, violations: list[dict]) 

50 每個 violation: {"rule": str, "severity": "warning"|"reject", "detail": str} 

51 """ 

52 violations = [] 

53 p_len = len(prompt_text) 

54 

55 # --- Rule 1: 字數限制 --- 

56 if p_len > 700: 

57 violations.append({ 

58 "rule": "R01_LENGTH_REJECT", 

59 "severity": "reject", 

60 "detail": f"提示詞字數 {p_len} 超過 700 字硬性上限。" 

61 }) 

62 elif p_len > 550: 

63 violations.append({ 

64 "rule": "R01_LENGTH_WARNING", 

65 "severity": "warning", 

66 "detail": f"提示詞字數 {p_len} 超過 550 字警告線。建議精簡。" 

67 }) 

68 

69 if p_len < 100: 

70 violations.append({ 

71 "rule": "R01_LENGTH_TOO_SHORT", 

72 "severity": "reject", 

73 "detail": f"提示詞字數 {p_len} 低於 100 字，內容不足。" 

74 }) 

75 

76 # --- Rule 2: 要求輸出審題過程但未禁止 --- 

77 if re.search(r"(先分析題型|先判斷題目類型|先進行審題)", prompt_text): 

78 if not re.search(r"(不要輸出審題|不得輸出審題|不輸出分析過程|嚴禁輸出.*審題)", prompt_text): 

79 violations.append({ 

80 "rule": "R02_ANALYSIS_LEAK", 

81 "severity": "reject", 

82 "detail": "包含「先分析題型」但未設定「不要輸出審題過程」，會導致答案前面出現廢話。" 

83 }) 

84 

85 # --- Rule 3: 反問防範 --- 

86 if not re.search(r"(不得反問|禁止反問|不要反問|不得提問|不得摻雜.*提問|禁止.*提問)", prompt_text): 

87 violations.append({ 

88 "rule": "R03_NO_ANTI_QUESTION", 

89 "severity": "warning", 

90 "detail": "未設定反問禁止語句，模型可能在答案中反問使用者。" 

91 }) 

92 

93 # --- Rule 4: 編造防範 --- 

94 if not re.search(r"(不得編造|嚴禁編造|嚴禁捏造|不要編造|禁止編造|杜絕捏造|嚴禁憑空捏造|不得.*捏造)", prompt_text): 

95 violations.append({ 

96 "rule": "R04_NO_ANTI_FABRICATION", 

97 "severity": "reject", 

98 "detail": "未設定法條/判決/年份/統計的防編造宣告。" 

99 }) 

100 

101 # --- Rule 5: 專家角色宣告 --- 

102 if not re.search(r"(專家|閱卷委員|寫作名師|考官|老師|評分委員)", prompt_text): 

103 violations.append({ 

104 "rule": "R05_NO_EXPERT_ROLE", 

105 "severity": "reject", 

106 "detail": "缺乏國考專家角色設定宣告。" 

107 }) 

108 

109 # --- Rule 6: 比較基準要求 --- 

110 if not re.search(r"(比較基準|比較.*基準|橫向對比|對比.*基準|比對)", prompt_text): 

111 violations.append({ 

112 "rule": "R06_NO_COMPARE_CRITERIA", 

113 "severity": "warning", 

114 "detail": "未提及比較題需建立比較基準。" 

115 }) 

116 

117 # --- Rule 7: 法律題分流 --- 

118 if not re.search(r"(法理|法律.*案例|三段論|涵攝|大前提)", prompt_text): 

119 violations.append({ 

120 "rule": "R07_NO_LEGAL_DIFFERENTIATION", 

121 "severity": "warning", 

122 "detail": "未區分法律法理題與法律案例題的作答策略。" 

123 }) 

124 

125 # --- Rule 8: 直接輸出要求 --- 

126 if not re.search(r"(直接輸出|直接.*正文|不得摻雜.*說明|不得.*輸出.*前言)", prompt_text): 

127 violations.append({ 

128 "rule": "R08_NO_DIRECT_OUTPUT", 

129 "severity": "warning", 

130 "detail": "未要求模型直接輸出答案正文（可能會輸出前導廢話）。" 

131 }) 

132 

133 # --- 第二層語意驗證 (#15)：對 reject 級違規做語意等價詞檢查 --- 

134 filtered_violations = [] 

135 for v in violations: 

136 if v["severity"] == "reject" and _semantic_check(v["rule"], prompt_text): 

137 # 語意等價詞命中，降級為 warning 

138 v["detail"] += "（語意驗證通過：包含等價表達）" 

139 v["severity"] = "warning" 

140 filtered_violations.append(v) 

141 

142 # --- 判定 --- 

143 has_reject = any(v["severity"] == "reject" for v in filtered_violations) 

144 passed = not has_reject 

145 

146 return passed, filtered_violations

149def main(): 

150 if len(sys.argv) < 2: 

151 print("用法: python3 scripts/gatekeeper.py <prompt_file>") 

152 print("範例: python3 scripts/gatekeeper.py prompts/current.md") 

153 sys.exit(1)
