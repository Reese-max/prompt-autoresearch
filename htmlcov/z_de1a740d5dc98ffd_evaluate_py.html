60def parse_judge_json(judge_output): 

61 json_match = re.search(r"```json\s*(.*?)\s*```", judge_output, re.DOTALL) 

62 if json_match: 

63 judge_json_str = json_match.group(1) 

64 else: 

65 judge_json_str = judge_output.strip() 

66 

67 if not judge_json_str.startswith("{"): 

68 start_idx = judge_json_str.find("{") 

69 end_idx = judge_json_str.rfind("}") 

70 if start_idx != -1 and end_idx != -1: 

71 judge_json_str = judge_json_str[start_idx:end_idx + 1] 

72 

73 return json.loads(judge_json_str, strict=False)

204def run_evaluation(prompt_file, question_file, max_workers=6, early_stop_threshold=None): 

205 """ 

206 運行完整題庫評估。 

207 early_stop_threshold: 若提供（baseline 平均分），跑完 1/3 題時若平均分低於此值 3 分以上則中止。 

208 """ 

209 # 載入所有依據檔案 

210 system_prompt = load_file(prompt_file) 

211 prompt_hash = sha256_text(system_prompt) 

212 general_rubric = load_file("rubrics/general.md") 

213 type_rubric = load_file("rubrics/type_specific.md") 

214 risk_rubric = load_file("rubrics/risk_rules.md") 

215 failure_taxonomy = load_file("rubrics/failure_taxonomy.md") 

216 

217 if not system_prompt: 

218 print(f"{C_RED}[錯誤] 找不到提示詞檔案: {prompt_file}{C_RESET}") 

219 sys.exit(1) 

220 

221 # 載入題庫 

222 questions = [] 

223 if not os.path.exists(question_file): 

224 print(f"{C_RED}[錯誤] 找不到題庫檔案: {question_file}{C_RESET}") 

225 sys.exit(1) 

226 

227 with open(question_file, "r", encoding="utf-8") as f: 

228 for line in f: 

229 if line.strip(): 

230 questions.append(json.loads(line.strip())) 

231 

232 total_q = len(questions) 

233 print(f"\n{C_PURPLE}開始評估題庫: {question_file} (共 {total_q} 題，並行執行緒: {max_workers}, prompt_hash={prompt_hash[:12]}){C_RESET}") 

234 

235 results = [] 

236 start_time = time.time() 

237 early_stop_at = total_q // 3 if early_stop_threshold else total_q # 1/3 處檢查 

238 early_stopped = False 

239 

240 # 使用 ThreadPoolExecutor 進行多執行緒並行呼叫 

241 with ThreadPoolExecutor(max_workers=max_workers) as executor: 

242 future_to_q = { 

243 executor.submit( 

244 evaluate_single_question, q, system_prompt, general_rubric, type_rubric, risk_rubric, failure_taxonomy, prompt_hash, question_file 

245 ): q for q in questions 

246 } 

247 

248 completed_count = 0 

249 for future in as_completed(future_to_q): 

250 q_obj = future_to_q[future] 

251 completed_count += 1 

252 try: 

253 res = future.result() 

254 results.append(res) 

255 # 即時在終端列印進度與單題得分 

256 err_flag = f" [ERROR: {res['error']}]" if "error" in res else "" 

257 fails_str = ", ".join(res.get("failures", [])) 

258 cache_flag = " [CACHE]" if res.get("cached") else "" 

259 print(f" [{completed_count}/{total_q}] 題號: {res['id']} ({res['type']}) -> 總分: {res['total_score']} (基礎: {res['general_score']}, 專項: {res['type_specific_score']}, 風險: {res['risk_score']}) | 缺陷: {fails_str or '無'}{err_flag}{cache_flag}") 

260 except Exception as e: 

261 print(f" [{completed_count}/{total_q}] 題號: {q_obj['id']} -> 執行異常: {str(e)}") 

262 results.append({ 

263 "id": q_obj["id"], "type": q_obj["type"], "question": q_obj["question"], 

264 "general_score": 0, "type_specific_score": 0, "risk_score": 0, "total_score": 0, 

265 "failures": ["F01"], "error": str(e) 

266 }) 

267 

268 # Early termination check (#10) 

269 if early_stop_threshold and completed_count >= early_stop_at and not early_stopped: 

270 interim_avg = sum(r.get("total_score", 0) for r in results) / len(results) 

271 if interim_avg < early_stop_threshold - 3.0: 

272 print(f"\n{C_YELLOW}[Early Stop] 前 {completed_count} 題平均分 {interim_avg:.2f} 低於 baseline {early_stop_threshold:.2f} 超過 3 分，中止評估。{C_RESET}") 

273 early_stopped = True 

274 # Cancel remaining futures 

275 for f in future_to_q: 

276 f.cancel() 

277 break 

278 

279 elapsed = time.time() - start_time 

280 print(f"\n{C_GREEN}評估完成！耗時: {elapsed:.2f} 秒 (約 {elapsed/60:.1f} 分鐘)。{C_RESET}") 

281 

282 # 3. 計算統計數據 

283 summary_stats = calculate_statistics(results, elapsed, prompt_file, question_file, prompt_hash) 

284 return summary_stats, results

286def calculate_statistics(results, elapsed_seconds, prompt_file, question_file, prompt_hash): 

287 """ 

288 計算評估總結指標。 

289 """ 

290 total_q = len(results) 

291 if total_q == 0: 

292 return {} 

293 

294 total_score = sum(r.get("total_score", 0) for r in results) 

295 avg_score = total_score / total_q 

296 

297 # 按題型計算平均分 

298 type_scores = {} 

299 type_counts = {} 

300 for r in results: 

301 t = r["type"] 

302 type_scores[t] = type_scores.get(t, 0) + r.get("total_score", 0) 

303 type_counts[t] = type_counts.get(t, 0) + 1 

304 

305 type_averages = {} 

306 for t in type_scores: 

307 type_averages[t] = type_scores[t] / type_counts[t] 

308 

309 # 統計缺陷頻率 

310 failure_counts = {} 

311 for r in results: 

312 for f in r.get("failures", []): 

313 if f: 

314 failure_counts[f] = failure_counts.get(f, 0) + 1 

315 

316 # 統計字數合格率 (800-1200字) 

317 in_range_count = sum(1 for r in results if 800 <= r.get("char_count", 0) <= 1200) 

318 word_count_pass_rate = (in_range_count / total_q) * 100 if total_q > 0 else 0 

319 

320 # 統計風險滿分率 (10分比例) 

321 risk_perfect_count = sum(1 for r in results if r.get("risk_score", 0) == 10) 

322 risk_perfect_rate = (risk_perfect_count / total_q) * 100 if total_q > 0 else 0 

323 cache_hit_count = sum(1 for r in results if r.get("cached")) 

324 error_count = sum(1 for r in results if r.get("error")) 

325 estimated_api_calls = (total_q - cache_hit_count) * 2 

326 

327 summary = { 

328 "timestamp": time.strftime("%Y-%m-%d %H:%M:%S"), 

329 "prompt_file": prompt_file, 

330 "question_file": normalize_path(question_file), 

331 "prompt_hash": prompt_hash, 

332 "elapsed_seconds": elapsed_seconds, 

333 "total_questions": total_q, 

334 "average_score": avg_score, 

335 "type_averages": type_averages, 

336 "failure_counts": failure_counts, 

337 "word_count_pass_rate": word_count_pass_rate, 

338 "risk_perfect_rate": risk_perfect_rate, 

339 "char_count": len(load_file(prompt_file)), 

340 "cache_hit_count": cache_hit_count, 

341 "error_count": error_count, 

342 "estimated_api_calls": estimated_api_calls, 

343 } 

344 

345 return summary
